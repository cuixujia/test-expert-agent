# AI生成测试用例工作空间目录结构
/home/instance/workspace/<userId>/<agentId>/
└── runtime_<projectId>/                      # 应用运行时根目录，所有会话共用，以雪狼项目为基准
    │
    ├── .opencode/                            # 运行时本地配置，用户自定义的智能体、工具、技能
    │   ├── agent/                            # Agent 定义、角色和权限
    │   ├── tool/                             # 自定义工具
    │   └── skills/                           # 技能、模板和引用资源
    │
    └── assets/                                   # 创建工作空间时默认创建
        └── <prdId>/                              # 需求目录，同需求会话共享
            └── prd.md                            # 【文件软链接】，需求文档
            └── <functionId>/                     # 业务功能目录
                ├── develop-docs/                 # 开发设计文档
                │   ├── fe-docs/                  # 【目录软链接】，前端开发设计文档。以迭代粒度沉淀，通过业务功能所属迭代关系获取
                │   │   ├── tech1.md
                │   │   └── tech2.md
                │   └── be-docs/                  # 【目录软链接】，后端开发设计文档。以迭代粒度沉淀，通过业务功能所属迭代关系获取
                │       ├── tech1.md
                │       └── tech2.md
                │
                ├── db-docs/                      # 【目录软链接】数据库资料，只读
                │   ├── tc-ddl.json               # 用例库资料，以业务功能粒度召回
                │   └── dict-ddl.json             # 数标库资料
                │
                ├── domain-knowledge/             # 【目录软链接】自定义领域知识，以“业务功能”对应的“业务领域”沉淀领域知识并自动召回
                │   ├── domain1.md                 
                │   └── domain1.md             
                │
                ├── code/                         # 【目录软链接】代码工程
                │   └── <源码文件及目录>
                │
                ├── output/                       # 【本地真实目录】用户确认采纳后的最终产物
                │   └── bf-tp-tc.json             # 测试用例集，业务功能-测试点-测试用例
                │
                └── <uuid>/                       # 本地真实目录：本次会话过程资料
                    ├── input/                    # 本次会话附带的输入资料
                    ├── prd-check.md              # 需求质检报告
                    ├── test-point.md             # 测试点清单
                    ├── tp-check.md               # 测试点质检报告
                    ├── design-check.md           # 详设质检报告
                    ├── test-case.md              # 测试用例清单
                    └── tc-check.md               # 测试用例质检报告
