    └── file/                                 # 创建工作空间时默认创建
        └── <bizFunctionId>/                  # 业务功能目录        
            ├── domain-knowledge/             # 【目录软链接】业务领域的领域知识

            │
            ├── asset-library/                # 【目录软链接】业务功能资产集
            │   → /home/test-case/<bizDomainId>/<bizFunctionId>/asset-library/
            │
            ├── code/                         # 【目录软链接】项目涉及到的代码工程集（难以定位到业务功能对应的代码）

            ├── output/                       # 共享的最终设计产物，用户验收后同步到「物理存储空间」进行归档

                ├── input/                    # 本次流程附带的输入资料（用户上传，只读）
                ├── state.md                  # 功能流程状态登记
